//while loop
var i = 1
while i <= 5 {
    print(i)
    i += 1
}

//repeaat while loop
var number = 6
repeat {
    print(number)
    number += 1
} while number <= 5

//break 
for i in 1...10 {
    if i == 5 {
        break
    }
    print(i)
}

//continue
for i in 1...8{
    if i == 4{
       continue
  }
  print(i)
}
